async function convertViaCloudConvert(file, format, progressElement) { progressElement.style.width = '20%'; // 1. Ask Cloudflare for the Render URL const urlRes = await fetch(WORKER_URL + "/get-render-url"); const urlData = await urlRes.json(); const RENDER_URL = urlData.url; progressElement.style.width = '30%'; // 2. Talk DIRECTLY to Render (No Cloudflare timeout!) const jobRes = await fetch(`${RENDER_URL}/api/v2/jobs`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ tasks: { "import-file": { operation: "import/upload" }, "convert-file": { operation: "convert", input: "import-file", output_format: format }, "export-file": { operation: "export/url", input: "convert-file" } } }) }); const jobData = await jobRes.json(); if (!jobData || !jobData.data) throw new Error("Render server is waking up. Please try again in 60 seconds."); progressElement.style.width = '50%'; const uploadTask = jobData.data.tasks.find(t => t.name === "import-file"); // 3. Upload the file directly const formData = new FormData(); formData.append('file', file); if (uploadTask.result.form.parameters) { for (const key in uploadTask.result.form.parameters) { formData.append(key, uploadTask.result.form.parameters[key]); } } await fetch(uploadTask.result.form.url, { method: 'POST', body: formData }); progressElement.style.width = '75%'; // 4. Check status directly while (true) { const statusRes = await fetch(`${RENDER_URL}/api/v2/jobs/${jobData.data.id}`); const statusData = await statusRes.json(); if (statusData.data.status === 'finished') { const exportTask = statusData.data.tasks.find(t => t.name === "export-file"); return exportTask.result.files[0].url; } else if (statusData.data.status === 'error') { throw new Error("Conversion failed on the server."); } await new Promise(r => setTimeout(r, 2000)); } }
